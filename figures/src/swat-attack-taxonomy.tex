\newcommand{\attackpanel}[7]{%
  \begin{scope}[shift={(#1,#2)}]
    \draw[panel] (0,0) rectangle (68,37);
    \node[anchor=west,text=swatblue,font=\bfseries\fontsize{10}{12}\selectfont]
      at (3,31.5) {#3};
    \node[anchor=east,smalllabel] at (65,31.5) {#4 attacks};
    \node[anchor=west] at (3,25) {#5};
    \foreach \i in {1,...,6} {
      \pgfmathsetmacro{\xx}{3+(\i-1)*10.5}
      \draw[draw=black!20,fill=black!3] (\xx,12) rectangle (\xx+9,18);
      \node[smalllabel] at (\xx+4.5,15) {P\i};
    }
    \foreach \i in {#6} {
      \pgfmathsetmacro{\xx}{3+(\i-1)*10.5}
      \draw[draw=swatblue,fill=swatblue] (\xx,12) rectangle (\xx+9,18);
      \node[text=white,font=\bfseries\fontsize{7}{9}\selectfont] at (\xx+4.5,15) {P\i};
      \fill[swatblue] (\xx+2.5,7) rectangle (\xx+4.5,9);
      \ifnum#7=2
        \fill[swatblue] (\xx+5.5,7) rectangle (\xx+7.5,9);
      \fi
    }
  \end{scope}}
